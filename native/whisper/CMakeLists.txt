cmake_minimum_required(VERSION 3.24)
project(gen2_whisper LANGUAGES C CXX)
if(NOT WHISPER_SOURCE_DIR)
 message(FATAL_ERROR "Supply the pinned whisper.cpp source checkout")
endif()
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${WHISPER_SOURCE_DIR}" OUTPUT_VARIABLE REV OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
execute_process(COMMAND git status --porcelain WORKING_DIRECTORY "${WHISPER_SOURCE_DIR}" OUTPUT_VARIABLE DIRTY OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT REV STREQUAL "d1be6fde11ac6e0407606b4e42fe72d34add8037" OR DIRTY)
 message(FATAL_ERROR "Whisper source must be clean at the reviewed revision")
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_COREML OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_ACCELERATE OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH armv8-a CACHE STRING "" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper EXCLUDE_FROM_ALL)
add_library(gen2_whisper SHARED shim.cpp)
option(GEN2_WHISPER_FRAMEWORK "Package the private iOS runtime as a framework" OFF)
if(GEN2_WHISPER_FRAMEWORK)
 cmake_minimum_required(VERSION 3.31)
 if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
  message(FATAL_ERROR "The framework target requires an explicit iOS SDK")
 endif()
 set_target_properties(gen2_whisper PROPERTIES
  FRAMEWORK TRUE
  OUTPUT_NAME Gen2Whisper
  MACOSX_FRAMEWORK_BUNDLE_NAME Gen2Whisper
  MACOSX_FRAMEWORK_IDENTIFIER com.pnola.Gen2Whisper
  MACOSX_FRAMEWORK_BUNDLE_VERSION 1
  MACOSX_FRAMEWORK_SHORT_VERSION_STRING 1.0
  INSTALL_NAME_DIR "@rpath")
endif()
target_compile_features(gen2_whisper PRIVATE cxx_std_17)
target_link_libraries(gen2_whisper PRIVATE whisper)
if(APPLE)
 target_link_options(gen2_whisper PRIVATE "-Wl,-exported_symbol,_gen2_whisper_abi" "-Wl,-exported_symbol,_gen2_whisper_open" "-Wl,-exported_symbol,_gen2_whisper_open_tiny_v1" "-Wl,-exported_symbol,_gen2_whisper_close" "-Wl,-exported_symbol,_gen2_whisper_run" "-Wl,-exported_symbol,_gen2_whisper_segments" "-Wl,-exported_symbol,_gen2_whisper_language" "-Wl,-exported_symbol,_gen2_whisper_segment")
elseif(UNIX)
 target_link_options(gen2_whisper PRIVATE "-Wl,--exclude-libs,ALL" "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map")
endif()
